--[[
================================================================
  AI CHATBOT GUI  -  Roblox Executor Script
================================================================
  A self-contained AI chatbot window that runs at executor level.

  FEATURES
  - ScreenGui parented to CoreGui / gethui() with IgnoreGuiInset
  - Round "AI" toggle button pinned to the RIGHT edge, middle of screen
  - ChatGPT-style dark theme with a sidebar listing saved chats
  - Markdown rendering: **bold**, `inline code` and fenced code
    blocks with their own header + Copy button
  - Typewriter effect on replies, Stop button while generating,
    Regenerate + Copy under every reply
  - Settings panel (gear at top right): Endpoint / API Key / Model /
    Temperature / System Prompt  (any OpenAI-compatible API)
  - Saved chats persist through writefile (falls back to memory
    automatically if the executor has no file functions)

  QUICK START
  1. Execute this script. Click the green "AI" button (right middle).
  2. Open Settings (gear, top right) and fill in your Endpoint + API key.
  3. Type a message and press Enter / Send.

  Works with: OpenAI, Groq, OpenRouter, local LLM servers, and any
  other OpenAI-compatible /v1/chat/completions endpoint.

  CONSOLE API
    getgenv().AIChatBot.Toggle()      getgenv().AIChatBot.Show()
    getgenv().AIChatBot.Hide()        getgenv().AIChatBot.OpenSettings()
    getgenv().AIChatBot.Unload()      getgenv().AIChatBotUnload()

  TIP: set FORCE_COREGUI = true below if your executor has no gethui()
  and you want the ScreenGui parented to game.CoreGui directly.
================================================================
]]

  if type(s) ~= "string" then return "" end
  return (s:gsub("^%s+", ""):gsub("%s+$", ""))
end

local orderCounter = 0

local function new(class, props, children)
  local inst = Instance.new(class)
  local parent = nil
  if props then
    for k, v in pairs(props) do
      if k == "Parent" then parent = v else inst[k] = v end
    end
  end
  if children then
    for _, c in ipairs(children) do c.Parent = inst end
  end
  if parent then inst.Parent = parent end
  return inst
end

local function readJSON(path)
  if not canWrite then return nil end
  if isFunc(isfile) and not isfile(path) then return nil end
  local ok, content = pcall(readfile, path)
  if not ok or type(content) ~= "string" or content == "" then return nil end
  local okD, data = pcall(function() return HttpService:JSONDecode(content) end)
  if not okD then return nil end
  return data
end

local function writeJSON(path, data)
  if not canWrite then return false end
  local ok, raw = pcall(function() return HttpService:JSONEncode(data) end)
  if not ok then return false end
  return (pcall(writefile, path, raw))
end

-- ================================ SAVED CHATS (data)
local sessions = {}
local activeId = nil

do
  local data = readJSON(CHAT_FILE)
  if type(data) == "table" and type(data.chats) == "table" then
    for _, c in ipairs(data.chats) do
      if type(c) == "table" and type(c.id) == "string" and type(c.messages) == "table" then
        table.insert(sessions, {
          id = c.id,
          title = (type(c.title) == "string" and c.title ~= "" and c.title) or "New Chat",
          messages = c.messages,
        })
      end
    end
  end
end

local function saveChats()
  writeJSON(CHAT_FILE, { version = 1, chats = sessions })
end

local function getSession(id)
  for _, s in ipairs(sessions) do
    if s.id == id then return s end
  end
  return nil
end

local function newSessionObject(title)
  local s = {
    id = HttpService:GenerateGUID(false),
    title = title or "New Chat",
    messages = {},
  }
  table.insert(sessions, 1, s)
  return s
end

local function escapeRich(s)
  s = s:gsub("&", "&amp;")
  s = s:gsub("<", "&lt;")
  s = s:gsub(">", "&gt;")
  return s

local function richifyText(raw)
  local s = (raw or ""):gsub("\r\n", "\n")
  s = escapeRich(s)
  s = s:gsub("`([^`\n]-)`", '<font name="Code">%1</font>')
  s = s:gsub("%*%*(.-)%*%*", "<b>%1</b>")
  local lines = {}
  for line in (s .. "\n"):gmatch("(.-)\n") do
    local l = line
    if l:match("^#+%s") or l:match("^#+$") then
      l = "<b>" .. l:gsub("^#+%s*", "") .. "</b>"
    else
      l = l:gsub("^%-+%s+", "•  ")
    end
    table.insert(lines, l)
  end
  return table.concat(lines, "\n")
end

-- split a message into normal text segments and fenced code blocks
local function parseSegments(text)
  local segs = {}
  local rest = text or ""
  while #rest > 0 do
    local s = string.find(rest, "```", 1, true)
    if not s then
      table.insert(segs, { type = "text", content = rest })
      break
    end
    local before = string.sub(rest, 1, s - 1)
    if trim(before) ~= "" then
      table.insert(segs, { type = "text", content = before })
    end
    local after = string.sub(rest, s + 3)
    local lang = ""
    local codeStart = 1
    local nl = string.find(after, "\n", 1, true)
    if nl then
      local first = trim(string.sub(after, 1, nl - 1))
      if first ~= "" and #first <= 20 and not string.find(first, "%s") then
        lang = first
        codeStart = nl + 1
      elseif first == "" then
        -- bare fence (no language line): skip the newline right after ```
        codeStart = nl + 1
      end
    end
    local e = string.find(after, "```", codeStart, true)
    local code
    if e then
      code = string.sub(after, codeStart, e - 1)
      rest = string.sub(after, e + 3)
    else
      code = string.sub(after, codeStart)
      rest = ""
    end
    table.insert(segs, { type = "code", lang = lang, content = (code:gsub("\n+$", "")) })
  end
  return segs
end

-- ================================ TYPEWRITER
local function typewrite(label, full, myGen, onDone)
  task.spawn(function()
    local n = #full
    local i = 0
    local acc = 0
    local last = os.clock()
    while i < n do
      task.wait()
      if myGen ~= genId or not label.Parent then break end
      local now = os.clock()
      acc = acc + (now - last) * 800
      last = now
      -- guarantee progress: at least 1 character per frame even if the
      -- clock delta is 0, so the effect can never stall
      local add = math.max(1, math.floor(acc))
      acc = math.max(0, acc - add)
      i = math.min(n, i + add)
      label.Text = string.sub(full, 1, i)
      scrollBottom()
    end
    onDone()
  end)
end

-- ================================ AI REQUEST FLOW
local function buildHistory(s)
  local msgs = {}
  if trim(settings.system_prompt) ~= "" then
    table.insert(msgs, { role = "system", content = settings.system_prompt })
  end
  local from = math.max(1, #s.messages - 13)
  for i = from, #s.messages do
    local m = s.messages[i]
    if m.role == "user" or m.role == "assistant" then
      table.insert(msgs, { role = m.role, content = m.content })
    end
  end
  return msgs
end

local function callAPI(messages, myGen, onDone)
  task.spawn(function()
    if not httpFunc then
      onDone(nil, "no HTTP request function found in this executor")
      return
    end
    local url = trim(settings.endpoint)
    url = url:gsub("/+$", "")
    if url == "" then url = DEFAULTS.endpoint end
    if not string.find(url, "/chat/completions", 1, true) then
      url = url .. "/chat/completions"
    end
    local headers = { ["Content-Type"] = "application/json" }
    if trim(settings.api_key) ~= "" then
      headers["Authorization"] = "Bearer " .. trim(settings.api_key)
    end
    local payload = {
      model = settings.model,
      messages = messages,
      temperature = settings.temperature,
      stream = false,
    }
    local ok, res = pcall(function()
      return httpFunc({
        Url = url,
        Method = "POST",
        Headers = headers,
        Body = HttpService:JSONEncode(payload),
      })
    end)
    if myGen ~= genId then onDone(nil, "cancelled") return end
    if not ok then
      onDone(nil, tostring(res))
      return
    end
    if type(res) ~= "table" then
      onDone(nil, "unexpected response type: " .. type(res))
      return
    end
    local body = res.Body or res.body
    if res.Success == false and not body then
      onDone(nil, "HTTP " .. tostring(res.StatusCode or "error"))
      return
    end
    local okJ, data = pcall(function() return HttpService:JSONDecode(body or "") end)
    if not okJ or type(data) ~= "table" then
      onDone(nil, "invalid API response: " .. string.sub(tostring(body), 1, 140))
      return
    end
    if type(data.error) == "table" and data.error.message then
      onDone(nil, tostring(data.error.message))
      return
    end
    local content = nil
    if type(data.choices) == "table" and data.choices[1] then
      local c = data.choices[1]
      if type(c.message) == "table" then content = c.message.content end
      if content == nil and type(c.delta) == "table" then content = c.delta.content end
      if content == nil then content = c.text end
    end
    if type(content) ~= "string" or content == "" then
      onDone(nil, "empty response from API")
      return
    end
    onDone(content, nil)
  end)
end

setSendMode = function(mode)
  sendMode = mode
  if mode == "stop" then
    sendBtn.Text = "Stop"
    sendBtn.BackgroundColor3 = Color3.fromRGB(70, 70, 70)
  else
    sendBtn.Text = "Send"
    sendBtn.BackgroundColor3 = THEME.accent
  end
  updateRegenVisibility()
end

stopGeneration = function()
  if not generating then return end
  generating = false
  genId = genId + 1
  setSendMode("send")
end

local function startGeneration()
  local s = activeId and getSession(activeId)
  if not s then return end
  if generating then return end
  generating = true
  genId = genId + 1
  local myGen = genId
  setSendMode("stop")
  local thinking = addThinking()
  callAPI(buildHistory(s), myGen, function(content, err)
    thinking.stop()
    if myGen ~= genId then
      generating = false
      setSendMode("send")
      return
    end
    if err then
      generating = false
      setSendMode("send")
      addErrorBubble(err)
      return
    end
    table.insert(s.messages, { role = "assistant", content = content })
    addAIMessage(content, true, myGen, function()
      generating = false
      setSendMode("send")
      saveChats()
      refreshSidebar()
    end)
  end)
end

requestAI = function()
  startGeneration()
end

regenerate = function()
  if generating then return end
  local s = activeId and getSession(activeId)
  if not s then return end
  if #s.messages == 0 then return end
  if s.messages[#s.messages].role ~= "assistant" then return end
  table.remove(s.messages)
  local last = messageRoots[#messageRoots]
  if last and last.role == "assistant" then
    last.root:Destroy()
    table.remove(messageRoots)
  end
  updateRegenVisibility()
  saveChats()
  startGeneration()
end

sendMessage = function(text)
  if generating then return end
  text = trim(text)
  if text == "" then return end
  local s = activeId and getSession(activeId) or nil
  if not s then
    local title = text:gsub("%s+", " ")
    if utf8 and utf8.len and utf8.offset then
      local okL, len = pcall(utf8.len, title)
      if okL and len and len > 28 then
        local okO, cut = pcall(utf8.offset, title, 29)
        if okO and cut then title = string.sub(title, 1, cut - 1) .. "..." end
      end
    elseif #title > 30 then
      title = string.sub(title, 1, 30) .. "..."
    end
    s = newSessionObject(title)
    activeId = s.id
    refreshSidebar()
  end
  table.insert(s.messages, { role = "user", content = text })
  if welcomeRoot then welcomeRoot:Destroy() welcomeRoot = nil end
  addUserMessage(text)
  scrollBottomForce()
  saveChats()
  startGeneration()
end

newChatBtn.MouseButton1Click:Connect(function()
  if generating then stopGeneration() end
  activeId = nil
  renderSession()
  refreshSidebar()
  pcall(function() input:CaptureFocus() end)
end)

sendBtn.MouseButton1Click:Connect(function()
  if sendMode == "stop" then
    stopGeneration()
  else
    sendMessage(input.Text)
    pcall(function() input:CaptureFocus() end)
  end
end)

input.FocusLost:Connect(function(enterPressed)
  if enterPressed and sendMode == "send" then
    sendMessage(input.Text)
    task.defer(function() pcall(function() input:CaptureFocus() end) end)
  end
end)

local armed = false
clearAllBtn.MouseButton1Click:Connect(function()
  if not armed then
    armed = true
    clearAllBtn.Text = "Sure? Click again"
    clearAllBtn.BackgroundColor3 = Color3.fromRGB(88, 44, 47)
    task.delay(3, function()
      armed = false
      clearAllBtn.Text = "Clear All Chats"
      clearAllBtn.BackgroundColor3 = THEME.rowBase
    end)
    return
  end
  armed = false
  clearAllBtn.Text = "Clear All Chats"
  clearAllBtn.BackgroundColor3 = THEME.rowBase
  if generating then stopGeneration() end
  sessions = {}
  activeId = nil
  renderSession()
  refreshSidebar()
  saveChats()
end)

-- ================================ UNLOAD / PUBLIC API
local function unload()
  for _, c in ipairs(connections) do
    pcall(function() c:Disconnect() end)
  end
  connections = {}
  if gui then gui:Destroy() end
  genv.AIChatBot = nil
  genv.AIChatBotUnload = nil
end

genv.AIChatBotUnload = unload
genv.AIChatBot = {
  Toggle = toggleWindow,
  Show = function() setWindow(true) end,
  Hide = function() setWindow(false) end,
  OpenSettings = openSettings,
  Unload = unload,
}

table.insert(connections, Players.PlayerRemoving:Connect(function(pl)
  if pl == Players.LocalPlayer then
    pcall(saveChats)
    pcall(saveSettings)
  end
end))
